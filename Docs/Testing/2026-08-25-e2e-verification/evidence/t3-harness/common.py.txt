def db(code):
    """Run `code` inside a manage.py shell against the local Postgres; return stdout."""
    env = dict(_os.environ)
    env.update(_DB_ENV)
    with _tf.NamedTemporaryFile('w', suffix='.py', delete=False) as fh:
        fh.write(code)
        path = fh.name
    out = _sp.run(
        [f'{REPO}/.venv/bin/python', 'manage.py', 'shell', '-c',
         f"exec(open({path!r}).read())"],
        cwd=REPO, env=env, capture_output=True, text=True, timeout=300,
    )
    if out.returncode != 0:
        raise RuntimeError(f'db() failed: {out.stdout[-2000:]}\n{out.stderr[-3000:]}')
    txt = out.stdout
    # strip the shell banner and Django/PIL log noise (DEBUG/INFO lines the
    # console logger emits inside the shell when DEBUG=True)
    def _noise(ln):
        if 'objects imported automatically' in ln:
            return True
        head = ln[:9]
        if len(head) == 9 and head[2] == ':' and head[5] == ':' and head[:2].isdigit():
            return True
        return False
    lines = [ln for ln in txt.splitlines() if not _noise(ln)]
    return '\n'.join(lines).strip()
