from jokes.models import Format, AgeRating, Language, Joke, JokeSubmission, MediaAsset, Appeal, ContentReport, JokePack
from django.contrib.auth import get_user_model
U = get_user_model()
print('formats', list(Format.objects.values_list('slug', flat=True)))
print('age_ratings', list(AgeRating.objects.values_list('slug','min_age')))
print('languages', list(Language.objects.values_list('code', flat=True))[:5])
print('jokes total', Joke.all_objects.count(), 'live', Joke.objects.count())
print('tier counts', dict((t, Joke.objects.filter(content_tier=t).count()) for t in ('tier_1','tier_2','tier_3')))
print('users', U.objects.count())
print('media assets', MediaAsset.objects.count())
print('appeals', Appeal.objects.count(), 'reports', ContentReport.objects.count())
print('packs', JokePack.objects.count())
