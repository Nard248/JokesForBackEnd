def probe(label, url):
    b = jget(a.get(url, timeout=30))
    rs = b.get("results", b if isinstance(b, list) else [])
    if isinstance(b, dict) and 'results' not in b: rs = [b] if 'id' in b else []
    rs = [j for j in rs if isinstance(j, dict) and 'id' in j]
    unread = [j for j in rs if j.get('id') not in ids[:11]]
    tgt = unread[:3]
    print(f"  {label:34} n={len(rs):3} unread_sample=" +
          str([(j['id'], j.get('is_locked'), bool(j.get('punchline')), bool(j.get('text'))) for j in tgt]))
    return tgt
