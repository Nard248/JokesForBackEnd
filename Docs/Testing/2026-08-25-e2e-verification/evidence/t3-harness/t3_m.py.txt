def mkdraft(fmt):
    r = s.post(f"{API}/jokes/my-drafts/", json={"format": fmt}, timeout=30)
    assert r.status_code == 201, dump(r)
    return r.json()['id']

# ---------------------------------------------------------------- T3-M01
draft_id = mkdraft('oneliner')
create_keys = None
r_c = s.get(f"{API}/jokes/my-drafts/{draft_id}/", timeout=30)
create_keys = sorted(r_c.json().keys())

r_patch1 = s.patch(f"{API}/jokes/my-drafts/{draft_id}/",
                   json={"text": "T3 partial autosa"}, timeout=30)
r_patch2 = s.patch(f"{API}/jokes/my-drafts/{draft_id}/",
                   json={"text": "T3 autosave: a one-liner that made it all the way to review.",
                         "age_rating": "family-friendly"}, timeout=30)
r_submit = s.post(f"{API}/jokes/my-drafts/{draft_id}/submit/", json={}, timeout=30)
r_repatch = s.patch(f"{API}/jokes/my-drafts/{draft_id}/",
                    json={"text": "edited after submit"}, timeout=30)
r_resubmit = s.post(f"{API}/jokes/my-drafts/{draft_id}/submit/", json={}, timeout=30)

ok = (r_patch1.status_code == 200 and r_patch2.status_code == 200
      and r_submit.status_code == 200 and r_submit.json().get('status') == 'pending'
      and r_repatch.status_code == 400
      and r_resubmit.status_code == 400
      and r_get.json().get('status') == 'pending'
      and r_get.json().get('text') != 'edited after submit')
rec('T3-M01',
    'draft create → PATCH autosave → submit flips to pending and locks the draft',
    'GIVEN an authenticated creator, WHEN they POST /jokes/my-drafts/ {format}, PATCH content '
    'into it (autosave, partial payloads accepted) and POST .../submit/, THEN the submission '
    'flips to status "pending" and every further PATCH/submit is refused 400 '
    '"Can only edit drafts or rejected submissions."',
    'SENSIBLE', ok,
    f"create id={draft_id} keys={create_keys}; PATCH#1={r_patch1.status_code}; "
    f"PATCH#2={r_patch2.status_code}; submit={dump(r_submit,120)}; "
    f"re-PATCH={dump(r_repatch,140)}; re-submit={dump(r_resubmit,140)}; "
    f"GET.status={r_get.json().get('status')} text={r_get.json().get('text')!r}")

# ------------------------------------------------------------- T3-M01b (DEFECT)
# The draft PATCH serializer backfills `text` from setup+punchline / lines.
# For the three formats whose FORMAT_RULES forbid `text` (setup, anti, knock)
# that backfilled value makes the draft permanently un-submittable.
bf = {}
for fmt, payload in (
    ('setup', {"setup": "Why did the draft cross the road?", "punchline": "It never got there."}),
    ('anti',  {"setup": "Two creators walk into a bar.", "punchline": "Nothing happens."}),
    ('knock', {"lines": ["Knock, knock.", "Who's there?", "Olive.", "Olive who?"]}),
):
    did = mkdraft(fmt)
    p = s.patch(f"{API}/jokes/my-drafts/{did}/", json=payload, timeout=30)
    sub = s.post(f"{API}/jokes/my-drafts/{did}/submit/", json={}, timeout=30)
    stored = db(f"from jokes.models import JokeSubmission; "
                f"x=JokeSubmission.objects.get(pk={did}); print(repr(x.text), x.status)")
    bf[fmt] = (did, p.status_code, sub.status_code, str(jget(sub))[:160], stored)

# control: the direct POST /jokes/submit/ path (validation BEFORE backfill) works
r_direct = s.post(f"{API}/jokes/submit/", json={
    "format": "setup", "age_rating": "family-friendly", "language": "en",
    "setup": "Why does the direct submit endpoint work?",
    "punchline": "Because it validates before it backfills."}, timeout=30)

broken = all(v[2] == 400 and 'Not allowed for' in v[3] for v in bf.values())
rec('T3-M01b',
    'a setup/anti/knock draft edited through the drafts editor can NEVER be submitted',
    'GIVEN a draft in a format whose FORMAT_RULES forbid `text` (setup, anti, knock), WHEN the '
    'autosave PATCH runs, THEN JokeSubmissionCreateSerializer.validate() backfills '
    'submission.text from setup+punchline / lines, and the later POST .../submit/ re-validates '
    'that stored text against FORMAT_RULES and 400s "Not allowed for <format> format." '
    'forever — the creator cannot submit a complete, valid joke.',
    'DEFECT-EXPECTED', broken,
    ' || '.join(f'{k}: draft={v[0]} PATCH={v[1]} submit={v[2]} {v[3]} stored_text/status={v[4]}'
                for k, v in bf.items())
    + f' || control POST /jokes/submit/ (no backfill re-read) = {dump(r_direct,140)}',
    note='Affects exactly the 3 formats that forbid `text`. Not caught by jokes/tests.py '
         'because test_submit_complete_draft_pending builds the JokeSubmission through the ORM '
         '(text stays "") instead of through the PATCH serializer. The SPA creator editor uses '
         'this exact path (src/features/create/api.ts).')

def jpeg(w=800, h=600, color=(200, 30, 60)):
    buf = io.BytesIO()
    Image.new('RGB', (w, h), color).save(buf, format='JPEG', quality=90)
    buf.seek(0)
    return buf

r_up = s.post(f"{API}/media/uploads/",
              files={'file': ('t3.jpg', jpeg(), 'image/jpeg')},
              data={'kind': 'image'}, timeout=120)

asset_id = asset.get('id')
disk = db(f"""

          '|safesearch=', a.safesearch, '|phash=', a.phash, '|owner=', a.owner_id)
""")
