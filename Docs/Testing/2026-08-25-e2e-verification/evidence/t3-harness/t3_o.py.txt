def share(joke_id, ua, session=None):
    s = session or sess()
    return s.get(f"{BASE}/jokes/{joke_id}/share/", headers={'User-Agent': ua},
                 timeout=30, allow_redirects=False)

def meta(html, prop, attr='property'):
    m = re.search(r'<meta\s+%s="%s"\s+content="([^"]*)"' % (attr, re.escape(prop)), html)
    return m.group(1) if m else None

# ---------------------------------------------------------------- T3-O01
r_bot = share(SETUP_JOKE, BOT_UA)
r_browser = share(SETUP_JOKE, BROWSER_UA)
hb, hh = r_bot.text, r_browser.text
canonical = re.search(r'<link rel="canonical" href="([^"]*)"', hb)
ldm = re.search(r'<script type="application/ld\+json">(.*?)</script>', hb, re.S)
ld = json.loads(ldm.group(1)) if ldm else None
refresh = re.search(r'<meta http-equiv="refresh" content="0; url=([^"]*)"', hb)
jsredir = 'location.replace("http://localhost:5273/jokes/%d")' % SETUP_JOKE in hb
ok = (r_bot.status_code == 200 and r_browser.status_code == 200
      and meta(hb, 'og:title') and meta(hb, 'og:description')
      and meta(hb, 'twitter:card', 'name') == 'summary_large_image'
      and canonical and canonical.group(1) == f'http://localhost:5273/jokes/{SETUP_JOKE}'
      and ld and ld.get('@type') == 'CreativeWork'
      and refresh and jsredir
      and hb == hh)
rec('T3-O01',
    'the share page always renders crawler metadata and bounces humans client-side',
    'GIVEN a public tier_1 joke, WHEN /jokes/<id>/share/ is fetched, THEN a 200 HTML shell is '
    'returned carrying per-joke og:title/og:description/og:url/og:image, '
    'twitter:card=summary_large_image, <link rel=canonical> to the SPA joke URL and a '
    'schema.org CreativeWork JSON-LD block; the human bounce is a <meta http-equiv="refresh"> '
    'plus location.replace() rather than a 3xx. NOTE: jokes.views.joke_share_page does NOT '
    'branch on User-Agent — bot and browser get byte-identical responses; scrapers simply '
    'ignore the refresh/JS while browsers act on it.',
    'SENSIBLE', ok,
    f'bot UA -> {r_bot.status_code} ({len(hb)}B), browser UA -> {r_browser.status_code} '
    f'({len(hh)}B), identical={hb == hh} | og:title={meta(hb,"og:title")!r} | '
    f'og:description={meta(hb,"og:description")!r} | og:image={meta(hb,"og:image")!r} | '
    f'og:url={meta(hb,"og:url")!r} | twitter:card={meta(hb,"twitter:card","name")!r} | '
    f'canonical={canonical.group(1) if canonical else None!r} | json-ld={json.dumps(ld)} | '
    f'meta-refresh={refresh.group(1) if refresh else None!r} | js location.replace={jsredir}',
    note='No UA sniffing exists; "redirect for humans" is purely client-side. A crawler that '
         'executes JS (Googlebot does) will also follow it to the SPA.')

# ---------------------------------------------------------------- T3-O02
# (a) the fixed regression: the punchline must not be in the title/description/JSON-LD
in_og = PUNCH_LINE in (meta(hb, 'og:title') or '') or PUNCH_LINE in (meta(hb, 'og:description') or '')
in_ld = PUNCH_LINE in json.dumps(ld or {})

# (b) tier_2 share page for an anonymous requester
r_t2 = share(TIER2_JOKE, BOT_UA)

            and 'application/ld+json' not in t2 and '<p class="joke-text">' not in t2
            and 'noindex' in t2)
ok = (not in_og and not in_ld and not in_title_tag and t2_clean)
rec('T3-O02',
    'the punchline never reaches og:title / og:description / <title> / JSON-LD, and a tier-gated '
    'share page exposes no content at all',
    'GIVEN a two-part joke, WHEN its share page is scraped, THEN the teaser used for <title>, '
    'og:title, og:description, twitter:* and the JSON-LD name/headline is joke.setup ONLY — the '
    'punchline appears in none of them (the spoiler regression fixed in 56e4945 stays fixed). '
    'AND GIVEN a tier_2 (mature) joke fetched by an anonymous/minor requester, THEN '
    'share_redirect.html is rendered: robots=noindex, no og:description, no og:image, no '
    'JSON-LD, no joke text.',

    f'punchline in og:title/description = {in_og}; in <title> = {in_title_tag}; in JSON-LD = '
    f'{in_ld} | tier_2 joke {TIER2_JOKE} -> HTTP {r_t2.status_code}, len={len(t2)}, '
    f'og:description={meta(t2,"og:description")!r}, og:image={meta(t2,"og:image")!r}, '
    f'json-ld present={"application/ld+json" in t2}, robots={meta(t2,"robots","name")!r}')

# ---------------------------------------------------------------- T3-O04
r_map = sess().get(f"{BASE}/sitemap.xml", timeout=60)
root = ET.fromstring(r_map.content)
ns = '{http://www.sitemaps.org/schemas/sitemap/0.9}'
locs = [el.text for el in root.iter(f'{ns}loc')]
paths = [l.replace('http://localhost:5273', '') for l in locs]
gated = ['/library', '/onboarding', '/create', '/settings', '/profile', '/inbox', '/saved',
         '/admin', '/login', '/register', '/billing', '/appeals']
present_gated = [g for g in gated if any(p == g or p.startswith(g + '/') for p in paths)]
statics = [p for p in paths if not p.startswith(('/jokes/', '/creators/', '/packs/'))]
ok = (r_map.status_code == 200
      and r_map.headers.get('content-type', '').startswith('application/xml')
      and root.tag == f'{ns}urlset'
      and not present_gated
      and all(l.startswith('http://localhost:5273/') for l in locs)
      and f'http://localhost:5273/jokes/{SETUP_JOKE}' in locs
      and f'http://localhost:5273/jokes/{TIER2_JOKE}' not in locs
      and sorted(statics) == sorted(['/', '/daily', '/trending', '/privacy', '/terms',
                                     '/cookie-policy', '/childrens-privacy']))
rec('T3-O04',
    'GET /sitemap.xml is valid sitemap XML of crawlable FRONTEND routes with gated routes excluded',
    'GIVEN an anonymous crawler, WHEN it GETs /sitemap.xml, THEN 200 application/xml with a '
    'sitemaps.org <urlset>; every <loc> is an absolute FRONTEND_URL route (never the backend '
    'host); the static section is exactly the 7 public marketing/legal routes; jokes are the '
    'tier_1 non-removed set an anonymous API caller can actually fetch (tier_2 excluded); '

    'gated route (/library, /onboarding, /create, /settings, /inbox, ...) appears.',
    'SENSIBLE', ok,
    f'HTTP {r_map.status_code} ct={r_map.headers.get("content-type")} bytes={len(r_map.content)} '
    f'| urls={len(locs)} | statics={sorted(statics)} | jokes={sum(1 for p in paths if p.startswith("/jokes/"))} '
    f'| creators={sum(1 for p in paths if p.startswith("/creators/"))} '
    f'| packs={sum(1 for p in paths if p.startswith("/packs/"))} '
