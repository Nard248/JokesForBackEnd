def make_user(tag):
    """Create a verified, active user directly and return a logged-in session."""
    email = f"{tag}{TS}@test.local"
    shell(f"""

U=G(); u,_=U.objects.get_or_create(username='{email}', defaults=dict(email='{email}'))
u.email='{email}'; u.is_active=True; u.set_password('{PW}'); u.save(); print(u.id)
""")
    s = sess(); csrf(s)
    r = s.post(f"{API}/auth/login/", json={"email": email, "password": PW}, timeout=30)
    assert r.status_code == 200, (r.status_code, r.text[:200])
    csrf(s)
    return s, email
