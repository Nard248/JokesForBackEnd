probe = shell("""

for j in Joke.objects.filter(is_removed=False).exclude(punchline='').exclude(punchline__isnull=True)[:600]:

    ws=[w for w in re.findall(r'[A-Za-z]{7,}', j.punchline or '') if w.lower() not in other]

        print(j.id); print(ws[0]); break
else:
    print('NONE'); print('NONE')
""")
